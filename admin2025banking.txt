$ErrorActionPreference = "Stop"
function JsonCall {
    param([string]$Method, [string]$Url, [hashtable]$Headers = $null, [hashtable]$Body = $null)
    try {
        $bodyJson = $null
        if ($Body) { $bodyJson = $Body | ConvertTo-Json -Depth 6 }
        if ($bodyJson) {
            return Invoke-RestMethod -Uri $Url -Method $Method -Headers $Headers -Body $bodyJson -ContentType "application/json"
        } else {
            return Invoke-RestMethod -Uri $Url -Method $Method -Headers $Headers
        }
    } catch {
        if ($_.Exception.Response) {
            $code = $_.Exception.Response.StatusCode.value__
            $reader = New-Object System.IO.StreamReader($_.Exception.Response.GetResponseStream())
            $resp = $reader.ReadToEnd()
            Write-Host "❌ HTTP $code -> $Url" -ForegroundColor Red
            Write-Host $resp
        } else {
            Write-Host $_.Exception.Message -ForegroundColor Red
        }
        return $null
    }
}
function Login($base, $email, $pwd) {
    return JsonCall -Method Post -Url "$base/api/auth/login" -Body @{ email = $email; motDePasse = $pwd }
}

$ErrorActionPreference = "Stop"
function JsonCall {
  param([string]$Method, [string]$Url, [hashtable]$Headers = $null, [hashtable]$Body = $null)
  try {
    $bodyJson = $null; if ($Body) { $bodyJson = $Body | ConvertTo-Json -Depth 6 }
    if ($bodyJson) {
      return Invoke-RestMethod -Uri $Url -Method $Method -Headers $Headers -Body $bodyJson -ContentType "application/json"
    } else {
      return Invoke-RestMethod -Uri $Url -Method $Method -Headers $Headers
    }
  } catch {
    if ($_.Exception.Response) {
      $code = $_.Exception.Response.StatusCode.value__
      $reader = New-Object System.IO.StreamReader($_.Exception.Response.GetResponseStream())
      $resp = $reader.ReadToEnd()
      Write-Host "❌ HTTP $code -> $Url" -ForegroundColor Red
      Write-Host $resp
    } else { Write-Host $_.Exception.Message -ForegroundColor Red }
    return $null
  }
}

import { HttpInterceptorFn } from '@angular/common/http';
import { catchError } from 'rxjs/operators';
import { throwError } from 'rxjs';
import { inject } from '@angular/core';
import { Router } from '@angular/router';

const TOKEN_KEY = 'auth_token';

export const tokenInterceptor: HttpInterceptorFn = (req, next) => {
  const router = inject(Router);
  const isBrowser = typeof window !== 'undefined' && typeof localStorage !== 'undefined';

  console.log(`[TOKEN_INTERCEPTOR] Processing ${req.method} ${req.url}`);

  let token: string | null = null;

  if (isBrowser) {
    token = localStorage.getItem(TOKEN_KEY);
    console.log(`[TOKEN_INTERCEPTOR] Token status: ${token ? 'FOUND' : 'NOT_FOUND'}`);

    // Debug: afficher les premières lettres du token si présent
    if (token) {
      console.log(`[TOKEN_INTERCEPTOR] Token preview: ${token.substring(0, 10)}...`);
    }
  } else {
    console.log('[TOKEN_INTERCEPTOR] Not in browser environment, no token available');
  }

  // Cloner la requête avec le token si disponible
  let authReq = req;
  if (token) {
    authReq = req.clone({
      setHeaders: {
        Authorization: `Bearer ${token}`
      }
    });
    console.log(`[HTTP DEBUG] ${authReq.method} ${authReq.url} AuthHeader: true`);
  } else {
    console.log(`[HTTP DEBUG] ${req.method} ${req.url} AuthHeader: false`);
  }

  return next(authReq).pipe(
    catchError(error => {
      console.error(`[TOKEN_INTERCEPTOR] HTTP error for ${req.method} ${req.url}:`, error);

      if (error.status === 401) {
        console.warn('[TOKEN_INTERCEPTOR] Unauthorized (401) - clearing token and redirecting');

        // Nettoyer le token invalide
        if (isBrowser) {
          const oldToken = localStorage.getItem(TOKEN_KEY);
          if (oldToken) {
            console.log('[TOKEN_INTERCEPTOR] Removing invalid token from localStorage');
            localStorage.removeItem(TOKEN_KEY);
          }
        }

        // Rediriger vers la page d'authentification
        console.log('[TOKEN_INTERCEPTOR] Redirecting to /auth');
        router.navigate(['/auth']);
      } else if (error.status === 403) {
        console.warn('[TOKEN_INTERCEPTOR] Forbidden (403) - insufficient permissions');
      } else if (error.status === 500) {
        console.error('[TOKEN_INTERCEPTOR] Internal Server Error (500)');
      }

      return throwError(() => error);
    })
  );
};

    let adminToken = '';
    let clientId: any = '';
    let numCompte = '';

    console.log('=== FLUX COMPLET INSCRIPTION ===');

    // 1) Admin login
    this.http.post<any>(`${base}/api/auth/login`, { email: adminEmail, motDePasse: adminPwd }).pipe(
      switchMap(res => {
        adminToken = res.data.token;
        console.log('✅ Admin connecté');
        return this.http.post<any>(`${base}/api/auth/register`, { prenom, nom, email, motDePasse });
      }),
      switchMap(res => {
        clientId = res.data?.id;
        console.log('✅ Client inscrit, id=', clientId);
        return this.http.post<any>(`${base}/api/auth/login`, { email, motDePasse });
      }),
      switchMap(res => {
        userToken = res.data.token;
        console.log('✅ Client connecté');
        return this.http.post<any>(`${base}/api/comptes`,
          { intitule: 'Compte principal', devise: 'EUR' },
          { headers: { Authorization: `Bearer ${userToken}` } }
        );
      }),
      switchMap(res => {
        numCompte = res.numCompte;
        console.log('✅ Compte créé:', numCompte);
        return this.http.post<any>(`${base}/api/comptes/${numCompte}/activer`, {},
          { headers: { Authorization: `Bearer ${adminToken}` } }
        );
      }),
      switchMap(() => {
        console.log('✅ Compte activé');
        return this.http.post<any>(`${base}/api/operations/deposit`,
          { numCompte, montant: 200, description: 'DEPOT_INIT' },
          { headers: { Authorization: `Bearer ${userToken}` } }
        );
      }),
      // 🎴 Émission carte (OPTIONNELLE - si ça échoue on continue)
      switchMap(() => {
        console.log('✅ Dépôt 200€ OK');
        return this.http.post<any>(`${base}/api/cards/${numCompte}/issue`, {},
          { headers: { Authorization: `Bearer ${adminToken}` } }
        ).pipe(
          catchError(err => {
            console.warn('⚠️ Émission carte ignorée (endpoint absent):', err.status);
            return of(null);
          })
        );
      }),
      // 🎴 Déblocage carte (OPTIONNEL - si ça échoue on continue)
      switchMap(() => {
        return this.http.post<any>(`${base}/api/cards/${numCompte}/debloquer`, {},
          { headers: { Authorization: `Bearer ${adminToken}` } }
        ).pipe(
          catchError(err => {
            console.warn('⚠️ Déblocage carte ignoré:', err.status);
            return of(null);
          })
        );
      }),
      finalize(() => this.loading.set(false))
    ).subscribe({
      next: () => {
        console.log('🎉 Inscription complète terminée !');
        this.auth.logout();
        this.hideSignupForm.set(true);
        this.showInscriptionSuccess.set(true);
        this.loginForm.patchValue({ email: email! });
        this.goToLogin();
        setTimeout(() => this.showInscriptionSuccess.set(false), 8000);
      },
      error: (err) => {
        console.error('=== ERREUR INSCRIPTION ===');
        console.error('URL:', err.url);
        console.error('Status HTTP:', err.status);
        console.error('Message backend:', err.error);

        let errorMessage = this.msg(err);

        if (err.status === 403) {
          errorMessage = 'Accès refusé : ' + (err.url?.includes('/activer') ? 'activation du compte impossible' : 'permissions insuffisantes');
        } else if (err.status === 400) {
          const backendMsg = err?.error?.message || err?.error?.error || err?.error;
          if (typeof backendMsg === 'string') {
            if (backendMsg.toLowerCase().includes('email')) {
              errorMessage = 'Cet email est déjà utilisé';
            } else if (backendMsg.toLowerCase().includes('password') || backendMsg.toLowerCase().includes('mot de passe')) {
              errorMessage = 'Le mot de passe ne respecte pas les critères requis';
            } else {
              errorMessage = backendMsg;
            }
          }
        }

        this.errorSignup.set(errorMessage);
      }
    });
  }

  private msg(err: any): string {
    if (err.status === 400) {
      const backendMsg = err?.error?.message || err?.error?.error;
      if (backendMsg) {
        return backendMsg;
      }
      return 'Données invalides. Vérifiez vos informations.';
    }
    return err?.error?.message ?? err?.message ?? 'Erreur réseau';
  }
